
\section{Introduction}
Given a functional $\calF:\calP(\R^d)\to\R$, its gradient-flow dynamics depend on the choice of geometry on the underlying space of measures. One of the most commonly used geometries is the Wasserstein-2 geometry, induced by the Wasserstein-2 distance. Its associated Onsager operator, which is the inverse of the Riemannian metric tensor, is given by
$\mathbb{K}_W(\rho): T_\rho^* \mathcal{M}^{+} \rightarrow T_\rho \mathcal{M}^{+}, \xi \mapsto-\operatorname{div}(\rho \nabla \xi)$.
Hence, the Wasserstein gradient flow of $\calF$ is
\begin{align}
\partial_t \mu_t = \nabla\cdot \left(\mu_t \nabla \calF'[\mu_t]\right). \tag{W-GF}
\end{align}
Similarly, the Fisher--Rao geometry is induced by the Hellinger distance, with associated Onsager operator
$\mathbb{K}_{\mathrm{FR}}(\rho): T_\rho^*\mathcal M^+\rightarrow T_\rho\mathcal M^+, \xi\mapsto \rho , \xi$.
Consequently, the Fisher--Rao gradient flow of $\calF$ is
\begin{align}
\partial_t\mu_t = -\mu_t \left( \calF'[\mu_t] - \int \calF'[\mu_t]  \dd \mu_t \right). \tag{FR-GF}
\end{align}
An interpolation between these two dynamics is given by the well-known Wasserstein--Fisher--Rao gradient flow:
\begin{align}
\partial_t\mu_t = \alpha \nabla\cdot \left(\mu_t \nabla \calF'[\mu_t]\right) - \beta \mu_t \left( \calF'[\mu_t] - \int \calF'[\mu_t]  \dd \mu_t \right). \tag{WFR-GF}
\end{align} 
Recently, a different geometry is considered in \cite{zhu2024inclusive,zhu2024kernel,gladin2024interaction}, which uses the MMD as the geometry. The associated Onsager operator is $\mathbb{K}_{\mathrm{MMD}}(\rho) = \mathbb{K}_{\mathrm{FR}}(\rho) \circ \mathcal{K}_\rho^{-1}$. The Interaction-Force Transport (IFT) gradient flow is described as the following:
\begin{align}
\partial_t\mu_t = -\mathbb{K}_{\mathrm{MMD}}(\mu_t) \calF'[\mu_t] = -\mu_t \mathcal{K}_{\mu_t}^{-1} \calF'[\mu_t] = - \mathcal{K}^{-1} \calF'[\mu_t]. \tag{IFT-GF}
\end{align}
Similarly as WFR-GF, an interpolation of WGF and IFT-GF can also be constructed
\begin{align*}
    \partial_t\mu_t = \alpha \nabla\cdot \left(\mu_t \nabla \calF'[\mu_t]\right) - \beta \mathcal{K}^{-1} \calF'[\mu_t] \tag{WIFT-GF}.
\end{align*}


\section{$\calF=\mmd$ and geometry=$\mmd$}
We focus on MMD due to its interpretation as the worse-case integration error in the unit ball of a RKHS. 
If $\calF$ is the MMD distance $\calF(\cdot)=\mmd^2(\cdot, \pi)$, then $\calF'[\mu](\cdot)=\int k(x, \cdot) \dd(\mu-\pi) = \mathcal{K}(\mu-\pi)$. 
MSIP uses the WFG-GF, but here we focus on the WIFT-GF. 
The main reason is that, \cite{gladin2024interaction} has proved that $\operatorname{MMD}^2\left(\mu_t, \pi\right) \leq e^{-2 \beta t} \cdot \operatorname{MMD}^2 \left(\mu_0, \pi\right)$. 

Following \cite{gladin2024interaction}, a way to simulate the WIFT-GF.  is via an weighted interacting particle system.  
Let $\mu_N^{(t)}=\sum_{i=1}^N w_i^{(t)} \delta_{x_i^{(t)}}$ be a weighted empirical measure at iteration $t$. Write $\bm{X}_t=(x_1^{(t)},\ldots,x_N^{(t)})$ and $\bm{w}^{(t)}=(w_1^{(t)},\ldots,w_N^{(t)})$.
Denote $m_\pi=\int k(x,\cdot)\dd \pi$ as the kernel mean embedding. 
First evolve the Wasserstein gradient term, 
\begin{align}\label{eq:wass_update}
    x_i^{(t+1)}=x_i^{(t)}-\tau \alpha\left[\sum_{j=1}^N w_j^{(t)} \nabla_2 k\left(x_j^{(t)}, x_i^{(t)}\right)-\nabla m_\pi\left(x_i^{(t)}\right)\right].
\end{align}
Then, evolve the IFT gradient term, by solving the following optimal weights: let $\eta=\beta\tau$, 
\begin{align}\label{eq:ift_update}
    \bm{w}^{(t+1)} = \arg \min _{\bm{w} \in \Delta^N} \left\{\frac{1}{2} \mmd^2\left(\sum_{i=1}^N w_i \delta_{x_i^{(t+1)}}, \pi\right)+\frac{1}{2 \eta} \mmd^2\left(\sum_{i=1}^N w_i \delta_{x_i^{(t+1)}}, \mu_N^{(t)}\right)\right\}.
\end{align}
Interestingly, letting $K^{(t+1)}=K(\bm{X}_{t+1})\in\R^{N\times N}$, the weight problem is
\begin{align*}
    \min_{\bm{w}\in \Delta^N} \left\{\frac{1}{2} \bm{w}^{\top} K^{(t+1)} \bm{w} - m_\pi(\bm{X}_{t+1})^{\top} \bm{w} + \frac{1}{2 \eta} \left(\bm{w}-\bm{w}^{(t)} \right)^{\top} K^{(t+1)} \left(\bm{w}-\bm{w}^{(t)}\right)\right\} ,
\end{align*}
By definition of kernel quadrature, there is $m_\pi(\bm{X}_{t+1}) = K^{(t+1)} \bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1})$.
And the solution is, when ignoring the simplex constraint, 
\begin{align*}
    \bm{w}^{(t+1)} = \frac{1}{1+\eta} \bm{w}^{(t)} + \frac{\eta}{1+\eta} \bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1}).
\end{align*}

In contrast, the update scheme for the standard Euclidean gradient is:
\begin{align}\label{eq:euc_update_1}
    x_i^{(t+1)}=x_i^{(t)}-\tau \alpha w_i \left[\sum_{j=1}^N w_j^{(t)} \nabla_2 k\left(x_j^{(t)}, x_i^{(t)}\right)-\nabla m_\pi\left(x_i^{(t)}\right)\right], \quad \bm{w}^{(t+1)} = \bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1}). 
\end{align}
Comparing the two update schemes, we note that the two only differs by a weight factor $w_i$. 

\begin{defi}[Weighted Euclidean Geometry]
For a positive weight vector $\bm{w}=(w_1,\ldots,w_N) \in\R_{>0}^N$, we equip the particle space $(\R^d)^N$ with the $\bm{w}$-weighted Euclidean inner product $\langle \bm{u},\bm{v}\rangle_{\bm{w}}
:= \sum_{i=1}^N w_i \langle u_i,v_i\rangle$, 
with the induced norm $\|\bm{u}\|_{\bm{w}}^2 := \sum_{i=1}^N w_i \|u_i\|_2^2$. 
For a differentiable function $f:(\R^d)^N\to\R$, its gradient with respect to the
$\bm{w}$-weighted Euclidean geometry, denoted by $\grad_{\bm{w}} f$, is defined through
\begin{align*}
    Df(\bm{X})[\bm{u}]
    =
    \left\langle \grad_{\bm{w}} f(\bm{X}),\bm{u}\right\rangle_{\bm{w}},
    \qquad
    \forall\,\bm{u}\in(\R^d)^N.
\end{align*}
Equivalently, its $i$-th component is $\left[\grad_{\bm{w}} f(\bm{X})\right]_i =
\frac{1}{w_i} \nabla_{x_i} f(\bm{X})$. 
\end{defi}


\begin{prop}[Smoothness under Euclidean and weighted Euclidean geometries]
Let $\bm{w}\in\Delta^N$ with $w_i>0$ for every $i$, and suppose that $k$ is symmetric and twice continuously differentiable, with $\sup_{x,y\in\R^d}\|\nabla_{11}^2 k(x,y)\|_{\mathrm{op}}<\infty$ and $\sup_{x,y\in\R^d}\|\nabla_{12}^2 k(x,y)\|_{\mathrm{op}}<\infty$. For fixed $\bm{w}$, consider
\[
E(\bm{X},\bm{w})
=
\frac{1}{2}\mmd^2\left(\sum_{i=1}^N w_i\delta_{x_i},\pi\right).
\]
Then $E(\cdot,\bm{w})$ is $L_{\mathrm{Euc}}$-smooth under the standard Euclidean geometry, with
\[
L_{\mathrm{Euc}}
\leq
\max_{1\leq i\leq N}w_i
\left(
2\sup_{x,y\in\R^d}\|\nabla_{11}^2 k(x,y)\|_{\mathrm{op}}
+
\sup_{x,y\in\R^d}\|\nabla_{12}^2 k(x,y)\|_{\mathrm{op}}
\right).
\]
Moreover, $E(\cdot,\bm{w})$ is $L_{\bm{w}}$-smooth under the $\bm{w}$-weighted Euclidean geometry, with
\[
L_{\bm{w}}
\leq
2\sup_{x,y\in\R^d}\|\nabla_{11}^2 k(x,y)\|_{\mathrm{op}}
+
\sup_{x,y\in\R^d}\|\nabla_{12}^2 k(x,y)\|_{\mathrm{op}}.
\]
\end{prop}

\begin{proof}
Recall that
\[
E(\bm{X},\bm{w})
=
\frac{1}{2}\sum_{i,j=1}^N w_iw_j k(x_i,x_j)
-
\sum_{i=1}^N w_i m_\pi(x_i)
+
\mathrm{const}.
\]
For any $\bm{u}=(u_1,\ldots,u_N)\in(\R^d)^N$, direct differentiation gives
\begin{align*}
\nabla_{\bm{X}}^2E(\bm{X},\bm{w})[\bm{u},\bm{u}]
={}&
\sum_{i,j=1}^N
w_iw_j
u_i^\top\nabla_{11}^2 k(x_i,x_j)u_i
\\
&+
\sum_{i,j=1}^N
w_iw_j
u_i^\top\nabla_{12}^2 k(x_i,x_j)u_j
-
\sum_{i=1}^N
w_i
u_i^\top\nabla^2m_\pi(x_i)u_i.
\end{align*}
Since $m_\pi(x)=\int k(x,y)\,\mathrm{d}\pi(y)$ and $\pi$ is a probability measure, we have
\[
\|\nabla^2m_\pi(x)\|_{\mathrm{op}}
=
\left\|
\int\nabla_{11}^2 k(x,y)\,\mathrm{d}\pi(y)
\right\|_{\mathrm{op}}
\leq
\sup_{x,y\in\R^d}
\|\nabla_{11}^2 k(x,y)\|_{\mathrm{op}}.
\]
Using $\sum_{j=1}^Nw_j=1$, it follows that
\[
\left|
\sum_{i,j=1}^N
w_iw_j
u_i^\top\nabla_{11}^2 k(x_i,x_j)u_i
-
\sum_{i=1}^N
w_i
u_i^\top\nabla^2m_\pi(x_i)u_i
\right|
\leq
2
\sup_{x,y\in\R^d}
\|\nabla_{11}^2 k(x,y)\|_{\mathrm{op}}
\sum_{i=1}^Nw_i\|u_i\|^2.
\]
For the mixed term,
\[
\left|
\sum_{i,j=1}^N
w_iw_j
u_i^\top\nabla_{12}^2 k(x_i,x_j)u_j
\right|
\leq
\sup_{x,y\in\R^d}
\|\nabla_{12}^2 k(x,y)\|_{\mathrm{op}}
\left(\sum_{i=1}^Nw_i\|u_i\|\right)^2.
\]
By Cauchy--Schwarz and $\sum_{i=1}^Nw_i=1$, we have
\[
\left(\sum_{i=1}^Nw_i\|u_i\|\right)^2
\leq
\sum_{i=1}^Nw_i\|u_i\|^2.
\]
Therefore,
\[
\left|
\nabla_{\bm{X}}^2 E(\bm{X},\bm{w})[\bm{u},\bm{u}]
\right|
\leq
\left(
2\sup_{x,y\in\R^d}\|\nabla_{11}^2 k(x,y)\|_{\mathrm{op}}
+
\sup_{x,y\in\R^d}\|\nabla_{12}^2 k(x,y)\|_{\mathrm{op}}
\right)
\sum_{i=1}^Nw_i\|u_i\|^2.
\]
Under the $\bm{w}$-weighted Euclidean geometry, $\|\bm{u}\|_{\bm{w}}^2=\sum_{i=1}^Nw_i\|u_i\|^2$. Hence
\[
\left|
\nabla_{\bm{X}}^2E(\bm{X},\bm{w})[\bm{u},\bm{u}]
\right|
\leq
L_{\bm{w}}\|\bm{u}\|_{\bm{w}}^2,
\]
where
\[
L_{\bm{w}}
=
2\sup_{x,y\in\R^d}\|\nabla_{11}^2 k(x,y)\|_{\mathrm{op}}
+
\sup_{x,y\in\R^d}\|\nabla_{12}^2 k(x,y)\|_{\mathrm{op}}.
\]
This proves smoothness under the weighted Euclidean geometry.

For the standard Euclidean geometry, since
\[
\sum_{i=1}^Nw_i\|u_i\|^2
\leq
\left(\max_{1\leq i\leq N}w_i\right)
\sum_{i=1}^N\|u_i\|^2
=
\left(\max_{1\leq i\leq N}w_i\right)
\|\bm{u}\|_2^2,
\]
we obtain
\[
\left|
\nabla_{\bm{X}}^2E(\bm{X},\bm{w})[\bm{u},\bm{u}]
\right|
\leq
L_{\mathrm{Euc}}\|\bm{u}\|_2^2,
\]
with
\[
L_{\mathrm{Euc}}
\leq
\max_{1\leq i\leq N}w_i
\left(
2\sup_{x,y\in\R^d}\|\nabla_{11}^2 k(x,y)\|_{\mathrm{op}}
+
\sup_{x,y\in\R^d}\|\nabla_{12}^2 k(x,y)\|_{\mathrm{op}}
\right).
\]
The proof is concluded.
\end{proof}


\subsection{Convergence}
\begin{theorem}[Monotone improvement]
\label{thm:wift-kq-descent}
At iteration $t\in\N$, let $\mu_N^{(t)}=\sum_{i=1}^N w_i^{(t)}\delta_{x_i^{(t)}}$ with $\bm{X}_t=(x_1^{(t)},\ldots,x_N^{(t)})$, and $\bm{w}^{(t)}=(w_1^{(t)},\ldots,w_N^{(t)})$. 

Assume that $k$ is twice continuously differentiable, that $K(\bm{X}_t)$ is positive definite for every $t$, and that the weights remain uniformly positive, namely $w_i^{(t)}\geq \underline w>0$ for every $i,t$. Assume moreover that, for every admissible $\bm{w}$, the map $\bm{X}\mapsto\mathcal{E}(\bm{X},\bm{w})$ has an $L_X$-Lipschitz gradient on a set containing all iterates. If $\tau\alpha\leq 2\underline w/L_X$, then for every $t\geq0$, 
\begin{align}
\operatorname{MMD} \left( \mu_N^{(t+1)},\pi
\right) \leq \operatorname{MMD}
\left( \mu_N^{(t)},\pi \right). 
\label{eq:wift-mmd-monotonicity}
\end{align} 
\end{theorem}

\begin{proof}
For fixed $\bm{X}$ and $\bm{w}$, let
$g_i(\bm{X},\bm{w})=\sum_{j=1}^N w_j\nabla_2 k(x_j,x_i)-\nabla m_\pi(x_i)$. 
Define the MMD energy by
\begin{align}\label{eq:defi_mmd}
\mathcal{E}(\bm{X},\bm{w})
=\frac12\operatorname{MMD}^2\left(\sum_{i=1}^N w_i\delta_{x_i},\pi\right),
\end{align}
Differentiating the MMD energy with respect to $x_i$ and using symmetry of the kernel gives
$\nabla_{x_i}\mathcal{E}(\bm{X},\bm{w})=w_i g_i(\bm{X},\bm{w})$.
Hence, the position update can be written as
$x_i^{(t+1)}=x_i^{(t)}-\frac{\tau\alpha}{w_i^{(t)}}\nabla_{x_i}\mathcal{E}(\bm{X}_t,\bm{w}^{(t)})$.
Thus, the Wasserstein substep is a diagonally preconditioned gradient step for $\bm{X}\mapsto\mathcal{E}(\bm{X},\bm{w}^{(t)})$. 
By the $L_X$-smoothness of $\mathcal{E}(\cdot,\bm{w}^{(t)})$, the descent lemma gives
\begin{align}
\mathcal{E}(\bm{X}_{t+1},\bm{w}^{(t)})
&\leq
\mathcal{E}(\bm{X}_t,\bm{w}^{(t)})
+
\sum_{i=1}^N
\left\langle
\nabla_{x_i}\mathcal{E}(\bm{X}_t,\bm{w}^{(t)}),
x_i^{(t+1)}-x_i^{(t)}
\right\rangle
+
\frac{L_X}{2}
\sum_{i=1}^N
\left\|
x_i^{(t+1)}-x_i^{(t)}
\right\|^2
\nonumber\\
&=
\mathcal{E}(\bm{X}_t,\bm{w}^{(t)})
-
\tau\alpha
\sum_{i=1}^N
\left(
w_i^{(t)}
-
\frac{L_X\tau\alpha}{2}
\right)
\left\|
g_i(\bm{X}_t,\bm{w}^{(t)})
\right\|^2.
\label{eq:wasserstein-descent}
\end{align}
Since $w_i^{(t)}\geq\underline w$ and $\tau\alpha\leq2\underline w/L_X$, we have
$w_i^{(t)}-L_X\tau\alpha/2\geq0$. Therefore,
$\mathcal{E}(\bm{X}_{t+1},\bm{w}^{(t)})\leq\mathcal{E}(\bm{X}_t,\bm{w}^{(t)})$. This proves that the particle update in Eq.~\eqref{eq:wass_update} is decreasing the MMD value. 

We next consider the weight update. For fixed $\bm{X}$, the dependence of the MMD energy on $\bm{w}$ is quadratic:
$\mathcal{E}(\bm{X},\bm{w})=\frac12 \bm{w}^\top K(\bm{X})\bm{w}-m_\pi(\bm{X})^\top \bm{w}+$const. Since $K(\bm{X})$ is positive definite, the unique minimizer is
$\bm{w}_{\mathrm{KQ}}(\bm{X})=K(\bm{X})^{-1}m_\pi(\bm{X})$.
Completing the square gives
\begin{align}
\mathcal{E}(\bm{X},\bm{w})
=
\mathcal{E}\left(\bm{X},\bm{w}_{\mathrm{KQ}}(\bm{X})\right)
+
\frac12 \left( \bm{w}-\bm{w}_{\mathrm{KQ}}(\bm{X}) \right)^\top K(\bm{X}) \left( \bm{w}-\bm{w}_{\mathrm{KQ}}(\bm{X}) \right).  
\label{eq:mmd-kq-decomposition}
\end{align}
By the weight update,
\begin{align}
\bm{w}^{(t+1)}
-
\bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1})
=
\frac{1}{1+\eta}
\left(
\bm{w}^{(t)}
-
\bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1})
\right).
\label{eq:weight-difference-contraction}
\end{align}
On the other hand, for fixed particle locations $\bm{X}_{t+1}$, the KQ decomposition gives
\begin{align}
&
\mathcal{E}(\bm{X}_{t+1},\bm{w}^{(t+1)})
-
\mathcal{E}\left(
\bm{X}_{t+1},
\bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1})
\right)
\nonumber\\
&\qquad=
\frac{1}{2}
\left(
\bm{w}^{(t+1)}
-
\bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1})
\right)^\top
K^{(t+1)}
\left(
\bm{w}^{(t+1)}
-
\bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1})
\right).
\label{eq:kq-energy-gap}
\end{align}
Substituting \eqref{eq:weight-difference-contraction} into
\eqref{eq:kq-energy-gap} yields
\begin{align}
&
\mathcal{E}(\bm{X}_{t+1},\bm{w}^{(t+1)})
-
\mathcal{E}\left(
\bm{X}_{t+1},
\bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1})
\right)
\nonumber\\
&\qquad=
\frac{1}{2(1+\eta)^2}
\left(
\bm{w}^{(t)}
-
\bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1})
\right)^\top
K^{(t+1)}
\left(
\bm{w}^{(t)}
-
\bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1})
\right)
\nonumber\\
&\qquad=
\frac{1}{(1+\eta)^2}
\left[
\mathcal{E}(\bm{X}_{t+1},\bm{w}^{(t)})
-
\mathcal{E}\left(
\bm{X}_{t+1},
\bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1})
\right)
\right].
\label{eq:kq-weight-contraction}
\end{align}
Since $(1+\eta)^{-2}\leq1$ and $\bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1})$ minimizes
$\bm{w}\mapsto\mathcal{E}(\bm{X}_{t+1},\bm{w})$, it follows that
$\mathcal{E}(\bm{X}_{t+1},\bm{w}^{(t+1)})
\leq
\mathcal{E}(\bm{X}_{t+1},\bm{w}^{(t)})$.
Combining the two substeps yields
\begin{align}
\mathcal{E}(\bm{X}_{t+1},\bm{w}^{(t+1)})
\leq
\mathcal{E}(\bm{X}_{t+1},\bm{w}^{(t)})
\leq
\mathcal{E}(\bm{X}_t,\bm{w}^{(t)}).
\label{eq:full-wift-descent}
\end{align}
Hence, $\mathcal{E}(\bm{X}_t,\bm{w}^{(t)})$ is nonincreasing, and the proof is concluded. 
\end{proof}


\begin{lem}[Uniform control of the kernel quadrature residual]
\label{lem:kq-residual-control}
Suppose that the assumptions of Theorem~\ref{thm:wift-kq-descent} hold and that the initial weights are chosen as $\bm{w}^{(0)}=\bm{w}_{\mathrm{KQ}}(\bm{X}_0) $.
Then, for every $t\geq 0$, $\operatorname{MMD}
\left(
\sum_{i=1}^N
w_{\mathrm{KQ},i} (\bm{X}_t)
\delta_{x_i^{(t)}},
\pi
\right)
\leq
\operatorname{MMD}
\left(
\sum_{i=1}^N
w_{\mathrm{KQ},i}(\bm{X}_0)
\delta_{x_i^{(0)}},
\pi
\right)$. 
\end{lem}

\begin{proof}
Recall the definition that $\mathcal{E}(\bm{X},\bm{w})
=\frac12\operatorname{MMD}^2\left(\sum_{i=1}^N w_i\delta_{x_i},\pi\right)$. For any fixed particle locations $\bm{X}_t$, the kernel quadrature weights
$\bm{w}_{\mathrm{KQ}}(\bm{X}_t)$ minimize the MMD energy over the weights. Hence, $\mathcal{E}\left(
\bm{X}_t,
\bm{w}_{\mathrm{KQ}}(\bm{X}_t)
\right)
\leq
\mathcal{E}\left(
\bm{X}_t,
\bm{w}^{(t)}
\right)$. On the other hand, by the monotonicity of the WIFT iteration established in
Theorem~\ref{thm:wift-kq-descent}, we have $\mathcal{E}\left(
\bm{X}_t,
\bm{w}^{(t)}
\right)
\leq
\mathcal{E}\left(
\bm{X}_0,
\bm{w}^{(0)}
\right)$. Since $\bm{w}^{(0)}=\bm{w}_{\mathrm{KQ}}(\bm{X}_0)$, we have
$ \mathcal{E}(\bm{X}_0,\bm{w}^{(0)})
=
\mathcal{E}(\bm{X}_0,\bm{w}_{\mathrm{KQ}}(\bm{X}_0)) $.
Combining the previous inequalities gives
\begin{align}
\mathcal{E}\left(
\bm{X}_t,
\bm{w}_{\mathrm{KQ}}(\bm{X}_t)
\right) \leq
\mathcal{E}\left(
\bm{X}_t,
\bm{w}^{(t)}
\right) \leq
\mathcal{E}\left(
\bm{X}_0,
\bm{w}^{(0)}
\right) = \mathcal{E}\left(
\bm{X}_0,
\bm{w}_{\mathrm{KQ}}(\bm{X}_0)
\right).
\end{align}
The proof is thus concluded. 
\end{proof}

\begin{theorem}[Convergence to the kernel quadrature error]
\label{thm:wift-convergence}
Suppose that the assumptions of Theorem~\ref{thm:wift-kq-descent} hold, and that the initial weights are chosen as $\bm{w}^{(0)}=\bm{w}_{\mathrm{KQ}}(\bm{X}_0)$. Then, the WIFT iterates satisfy
\begin{align*}
\operatorname{MMD}^2(\mu_N^{(t)},\pi) \leq
\frac{1}{(1+\eta)^{2t}}
\operatorname{MMD}^2(\mu_N^{(0)},\pi) +
\left(
1-\frac{1}{(1+\eta)^{2t}}
\right)
\operatorname{MMD}^2
\left(
\sum_{i=1}^N
w_{\mathrm{KQ},i}(\bm{X}_0)
\delta_{x_i^{(0)}},
\pi
\right). 
\end{align*}
\end{theorem}

\begin{proof}
From the exact contraction of the weight update in
Eq.~\eqref{eq:kq-weight-contraction}, together with descent of the Wasserstein position update, we have
\begin{align}
\mathcal{E}(\bm{X}_{t+1},\bm{w}^{(t+1)}) \leq \frac{1}{(1+\eta)^2}
\mathcal{E}(\bm{X}_t,\bm{w}^{(t)}) + \left(
1 - \frac{1}{(1+\eta)^2}
\right)
\mathcal{E} \left(
\bm{X}_{t+1},
\bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1}) \right).
\label{eq:wift-one-step-convergence}
\end{align}
By \Cref{lem:kq-residual-control}, 
$\mathcal{E}(\bm{X}_{t+1},\bm{w}_{\mathrm{KQ}}(\bm{X}_{t+1}))
\leq
\mathcal{E}(\bm{X}_0,\bm{w}_{\mathrm{KQ}}(\bm{X}_0))$,
and therefore
\begin{align}
\mathcal{E}(\bm{X}_{t+1},\bm{w}^{(t+1)}) \leq
\frac{1}{(1+\eta)^2}
\mathcal{E}(\bm{X}_t,\bm{w}^{(t)}) +
\left(
1-\frac{1}{(1+\eta)^2}
\right)
\mathcal{E}\left(
\bm{X}_0,
\bm{w}_{\mathrm{KQ}}(\bm{X}_0)
\right).
\label{eq:wift-recursion-initial-kq}
\end{align}
The proof is concluded. 
\end{proof}


\begin{rem}
Theorem~\ref{thm:wift-convergence} separates the optimization error from the
statistical error of the initial kernel quadrature rule. In particular, if
$\operatorname{MMD}(\sum_{i=1}^N
w_{\mathrm{KQ},i}(\bm{X}_0)\delta_{x_i^{(0)}},\pi)
=O_{\mathbb{P}}(N^{-s/d})$, then
\begin{align*}
\operatorname{MMD}(\mu_N^{(t)},\pi)
&\leq \frac{1}{(1+\eta)^t}
\operatorname{MMD}(\mu_N^{(0)},\pi)
+O_{\mathbb{P}}(N^{-s/d}).
\end{align*}
Thus, the optimization error decays geometrically until it reaches the
kernel quadrature error floor. If, in addition,
$\operatorname{MMD}(\mu_N^{(0)},\pi)=O_{\mathbb{P}}(1)$, then choosing
$t\geq \frac{s}{d\log(1+\eta)}\log N$ is sufficient to obtain
$\operatorname{MMD}(\mu_N^{(t)},\pi)=O_{\mathbb{P}}(N^{-s/d})$.
\end{rem}
